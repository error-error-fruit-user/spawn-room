// The game's options, added to the Options menu (Horde Hammer adds its own Horde Hammer Options)

OptionValue "SpawnRoom_Zombies" {
    0, "$SPAWNROOM_MENU_ZOMBIES_HH"
    1, "$SPAWNROOM_MENU_ZOMBIES_VARIED"
}

AddOptionMenu "OptionsMenu" {
    Option "$SPAWNROOM_MENU_ZOMBIES", "spawnroom_varied_zombies", "SpawnRoom_Zombies"
}

// and the simple options menu, which UZDoom shows instead while m_simpleoptions is on
AddOptionMenu "OptionsMenuSimple" {
    Option "$SPAWNROOM_MENU_ZOMBIES", "spawnroom_varied_zombies", "SpawnRoom_Zombies"
}
